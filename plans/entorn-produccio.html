2. scripts/build-prod.sh

Script que sobreescriu temporalment l'entry point i compila.

#!/bin/bash
# scripts/build-prod.sh
set -e

echo "🏭 Compilant bundle de producció..."

# Sobreescriu temporalment el main.jsx amb el de producció
cp src/main.jsx src/main.jsx.bak
cp src/main-prod.jsx src/main.jsx

# Build
npm run build

# Restaura
mv src/main.jsx.bak src/main.jsx
echo "✅ Bundle PROD generat a dist/"

3. package.json — script nou

Afegir "build:prod" al package.json.

{
  "scripts": {
    "dev": "vite",
    "build": "vite build",
    "build:prod": "VITE_PROD_MODE=true vite build",
    "preview": "vite preview"
  }
}
